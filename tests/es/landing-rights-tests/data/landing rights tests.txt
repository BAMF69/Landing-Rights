test "Landing Rights claim and ship"
	status active
	description "Buy the New Boston claim, then receive a shipment from Dyson Sphere Program."
	sequence
		inject "Landing Rights New Boston Save"
		call "Load First Savegame"
		# A shipment placed in the bridge folder before launch is delivered on landing (load lands you).
		input
			key "Return"
		assert
			"landing rights: claim New Boston" == 0
		call "Depart"
		navigate
			"travel destination" "New Boston"
		call "Land"
		# The claim offer: pick "I'll take it."
		input
			key 1
		input
			key "Return"
		assert
			"landing rights: claim New Boston" == 1
		# F on the planet screen sends you back down to the claim (re-announces the landing).
		input
			key f
		assert
			"flagship landed" == 1
		call "Depart"
		navigate
			"travel destination" "New Boston"
		call "Land"
		# Landing saves the game; the shipped Metal must be in the hold.
